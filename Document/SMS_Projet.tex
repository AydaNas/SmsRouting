\section*{Rules for Antenna}

\subsection*{Rule 1 — Antenna Discovery}

\[
\texttt{Channel\_in}\{\texttt{exchange : Search}\} ? \langle\texttt{Search,ID\_user, userLocationX, userLocationY}\rangle
\]
\[
\&\&\ \sqrt{(\texttt{userLocationX} - \texttt{antenna\_position\_x})^2 + (\texttt{userLocationY} - \texttt{antenna\_position\_y})^2} \leq \texttt{radius}
\]
\hspace*{3em}$\texttt{dist} = \sqrt{(\texttt{userLocationX} - \texttt{antenna\_position\_x})^2 + (\texttt{userLocationY} - \texttt{antenna\_position\_y})^2}$
\hspace*{3em}$\texttt{ChannelOutSet.get(ID\_user)}\ !\  \texttt{Search\_reply}\langle\texttt{ID\_antenna, dist}\rangle$

\subsection*{Rule 2 — Registration of users}

\[
\texttt{Channel\_in}\{\texttt{exchange : Connection}\}\ ?\ \langle\texttt{Connection,ID\_user}\rangle
\]
\[
\&\&\ \texttt{target\_antenna = ID\_antenna}
\]
\hspace*{3em}\texttt{Table\_User\_position[target\_antenna] = ID\_user}\\
\hspace*{3em}$\forall\ \texttt{n}\ \in \texttt{ ChannelOutSet }\ |\ \texttt{n is a channel of an antennna} $\\
\hspace*{6em}$\texttt{n}\{\texttt{exchange : Update}\}\ ?\ \texttt{Update\_Location}\langle\texttt{ID\_user,ID\_antenna}\rangle $

\subsection*{Rule 3 — SMS transmission}

\[
\texttt{Channel\_in}\ ?\ \langle\texttt{SMS, sender\_id , receiver\_id, message}\rangle
\]
\hspace*{3em}$\textbf{if}\ \texttt{reciver\_id} \notin \texttt{Table\_User\_position} \textbf{then}$\\
\hspace*{6em}\texttt{drop}\\
\hspace*{3em}$\textbf{else if}\ \texttt{Table\_User\_position[receiver\_id]}\ =\ \text{ID\_antenna}\ \textbf{then}$\\
\hspace*{6em}\texttt{user\_channel = ChannelOutSet.get(receiver\_id)}\\
\hspace*{6em}\texttt{user\_channel}\ $!$\ \texttt{$\langle$SMS,sender\_Id , receiver\_Id, message$\rangle$}\\
\hspace*{3em}\textbf{else}\\
\hspace*{6em}\texttt{antenna\_channel = ChannelOutSet.get(Table\_User\_position[receiver\_id])}\\
\hspace*{6em}\texttt{antenna\_channel}\ $!$\ \texttt{$\langle$SMS,sender\_Id , receiver\_Id, message$\rangle$}

\subsection*{Rule 4 — Update Location}

\[
\texttt{Channel\_in}\{\texttt{exchange : Update}\}\ ?\ \langle\texttt{Update\_Location, id\_user, id\_antenna}\rangle
\]
\[
\&\&(\ \texttt{id\_user}\ \notin\ \texttt{Table\_User\_position}\ ||\ \texttt{id\_antenna} = \texttt{Table\_User\_position[id\_user]})
\]
\hspace*{3em}$\texttt{Table\_User\_position[id\_user]} = \texttt{id\_antenna}$\\
 \hspace*{3em}$\forall\ \texttt{n}\ \in \texttt{ ChannelOutSet }\ |\ \texttt{ n is a channel of an antennna} $\\
\hspace*{6em}$\texttt{n}\{\texttt{exchange : Update}\}\ ?\ \texttt{Update\_Location}\langle\texttt{ID\_user,ID\_antenna}\rangle $

\section*{Rules for user}

\subsection*{Rule 1 — Search antenna (new connection)}

\[
\texttt{req}\ ?\ \texttt{Search\_antenna\_msg}
\]
\hspace*{3em}$\forall s \in \texttt{antennaChannelset}\ \texttt{do:}\\
\hspace*{6em}\texttt{s}\{\texttt{exchange : Search}\}$\ !\ \texttt{Search}$\langle$\texttt{ID\_user}$\rangle$

\subsection*{Rule 2 — Establish a connection}

\[
\texttt{Channel\_in}\ ?\ \langle\texttt{Search\_reply},\ \texttt{test},\ \texttt{distance}\rangle
\]
\hspace*{3em}\texttt{best\_antenna = select antenna(ID) with min(distance)}\\
\hspace*{3em}\texttt{channel\_best\_antenna = antennaChannelset.get(best\_antenna)}\\
\hspace*{3em}$\textbf{if}\ \texttt{best\_antenna} \neq \texttt{current\_antenna}\ \textbf{then}$\\
\hspace*{6em}\texttt{channel\_best\_antenna}\{\texttt{exchange : Connection}\}\ !\ \texttt{connection}$\langle$\texttt{ID\_user}$\rangle$\\
\hspace*{6em}\texttt{current\_Id\_antenna := best\_antenna}

\subsection*{Rule 3 — Move to another antenna}

\[
\texttt{req}\ ?\ \langle\texttt{Move},\ \texttt{x\_coordinate},\ \texttt{y\_coordinate}\rangle
\]
\hspace*{3em}\texttt{userLocationX = x\_coordinate}\\
\hspace*{3em}\texttt{userLocationY = y\_coordinate}\\
\hspace*{3em}$\forall s \in \texttt{antennaChannelset}\ \texttt{do:}\\
\hspace*{6em}\texttt{s}\{\texttt{exchange : Search}\}$\ !\ \texttt{Search}$\langle$\texttt{ID\_user}$\rangle$

\subsection*{Rule 4 — User send message}

\[
\texttt{req}\ ?\ \langle\texttt{SMS\_user\_itf},\ \texttt{ID\_user\_to},\ \texttt{message}\rangle
\]
\hspace*{3em}\texttt{receiver\_ID} $=$ \texttt{ID\_user\_to}\\
\hspace*{3em}\texttt{Channel\_out}\ !\ \texttt{SMS}$\langle$\texttt{sender\_Id},\ \texttt{receiver\_Id},\ \texttt{message}$\rangle$


\subsection*{Rule 5 — User receive message}

\[
\texttt{Channel\_in}\ ?\ \langle\texttt{User\_send\_message},\ \texttt{ID\_user\_from},\ \texttt{ID\_user\_to},\ \texttt{message}\rangle\ 
\]
\[
\&\&\ \texttt{current\_antenna = antenna\_id} 
\]
\hspace*{3em}\texttt{print message}

\subsection*{Rule 6 — User receive message}

\[
\texttt{req}\ ?\ \texttt{Leave}
\]
\hspace*{3em}\texttt{channel\_in.close()}\\
\hspace*{3em}\texttt{channel\_out.close()}\\
\hspace*{3em}\texttt{End of programme}
