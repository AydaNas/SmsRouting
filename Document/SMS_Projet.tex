\section*{Rules for Antenna}

\subsection*{Rule 1 — User connection (new user)}

\[
\texttt{Channel\_in}\ ?\ \langle\texttt{User\_connected},\ \texttt{ID\_user}\rangle
\]

\textbf{if} \texttt{ID\_user} $\notin$ \texttt{Table\_User} \textbf{then}\\
\hspace*{3em}\texttt{Table\_User.add(ID\_user,\ id\_user\_channel\_out)}\\
\hspace*{3em}\texttt{Table\_User\_position.add(ID\_user,\ ID\_antenna)}\\
\hspace*{3em}\textbf{for}\ \texttt{a}\ $\in$\ \texttt{Table\_Antenna}\\
\hspace*{6em}\texttt{a.Channel}\ $!$\ \texttt{Antenna\_update\_user}$\langle$\texttt{ID\_user},\ \texttt{ID\_antenna}$\rangle$\\
\hspace*{3em}\texttt{id\_user\_channel\_out}\ $!$\ \texttt{Antenna\_Ack}

\textbf{else}\\
\hspace*{3em}\texttt{id\_user\_channel\_out}\ $!$\ \texttt{Antenna\_Ack}


\subsection*{Rule 2 — User deconnection (exiting user)}

\[
\texttt{Channel\_in}\ ?\ \langle\texttt{User\_disconnected},\ \texttt{ID\_user}\rangle
\]

\textbf{if} \texttt{ID\_user} $\in$ \texttt{Table\_User} \textbf{then}\\
\hspace*{3em}\texttt{tmp\_channel = Table\_User.get(ID\_user)}\\
\hspace*{3em}\texttt{Table\_User.remove(ID\_user)}\\
\hspace*{3em}\texttt{Table\_User\_position.remove(ID\_user)}\\
\hspace*{3em}\textbf{for}\ \texttt{a}\ $\in$\ \texttt{Table\_Antenna}\\
\hspace*{6em}\texttt{a.Channel}\ $!$\ \texttt{Antenna\_delete\_user}$\langle$\texttt{ID\_user}$\rangle$

\subsection*{Rule 3 — User send message}

\[
\texttt{Channel\_in}\ ?\ \langle\texttt{User\_send\_message},\ \texttt{ID\_user\_from},\ \texttt{ID\_user\_to},\ \texttt{message}\rangle
\]
\hspace*{3em}\texttt{target\_antenna = Table\_User\_position.get(ID\_user\_to)}

\textbf{if} \texttt{target\_antenna = ID\_antenna} \textbf{then}\\
\hspace*{6em}\texttt{out\_channel = Table\_User.get(ID\_user\_to)}

\textbf{else}\\
\hspace*{6em}\texttt{next\_antenna = Routing\_Table.get(target\_antenna)}\\
\hspace*{6em}\texttt{out\_channel = Table\_Antenna.get(next\_antenna)}\\
\hspace*{3em}\texttt{out\_channel}\ !\ \text{User\_send\_message}$\langle$\texttt{ID\_user\_from},\ \texttt{ID\_user\_to},\ \texttt{message}$\rangle$

\subsection*{Rule 4 — Antenna update user}

\[
\texttt{Channel\_in}\ ?\ \langle\texttt{Antenna\_update\_user},\ \texttt{ID\_user},\ \texttt{ID\_antenna}\rangle
\]

\textbf{if} \texttt{ID\_user} $\in$ \texttt{Table\_User\_position} \textbf{and}\\\hspace*{3em}\texttt{ID\_antenna = Table\_User\_position.get(ID\_user)} \textbf{then}\\
\hspace*{3em}\texttt{Do nothing}

\textbf{else}\\
\hspace*{3em}\texttt{Table\_User\_position.set(ID\_user\,ID\_antenna)}\\
\hspace*{3em}\textbf{for}\ \texttt{a}\ $\in$\ \texttt{Table\_Antenna}\\
\hspace*{6em}\texttt{a.Channel}\ $!$\ \texttt{Antenna\_update\_user}$\langle$\texttt{ID\_user},\ \texttt{ID\_antenna}$\rangle$

\subsection*{Rule 5 — Antenna delete user}

\[
\texttt{Channel\_in}\ ?\ \langle\texttt{Antenna\_delete\_user},\ \texttt{ID\_user}\rangle
\]

\textbf{if} \texttt{ID\_user} $\notin$ \texttt{Table\_User\_position} \textbf{then}\\
\hspace*{3em}\texttt{Do nothing}

\textbf{else}\\
\hspace*{3em}\texttt{Table\_User\_position.delete(ID\_user)}\\
\hspace*{3em}\textbf{for}\ \texttt{a}\ $\in$\ \texttt{Table\_Antenna}\\
\hspace*{6em}\texttt{a.Channel}\ $!$\ \texttt{Antenna\_delete\_user}$\langle$\texttt{ID\_user}$\rangle$
